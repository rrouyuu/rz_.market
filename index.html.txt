        // 自動圖片極致壓縮機制
        function compressImage(file, maxWidth = 600, maxHeight = 600, quality = 0.5) {
            return new Promise((resolve) => {
                const reader = new FileReader();
                reader.readAsDataURL(file);
                reader.onload = (event) => {
                    const img = new Image();
                    img.src = event.target.result;
                    img.onload = () => {
                        let width = img.width;
                        let height = img.height;

                        if (width > height) {
                            if (width > maxWidth) {
                                height = Math.round((height * maxWidth) / width);
                                width = maxWidth;
                            }
                        } else {
                            if (height > maxHeight) {
                                width = Math.round((width * maxHeight) / height);
                                height = maxHeight;
                            }
                        }

                        const canvas = document.createElement('canvas');
                        canvas.width = width;
                        canvas.height = height;
                        const ctx = canvas.getContext('2d');
                        ctx.drawImage(img, 0, 0, width, height);

                        resolve(canvas.toDataURL('image/jpeg', quality));
                    };
                };
            });
        }

            e.preventDefault();

            const textUrls = document.getElementById('admin-p-img-urls').value.split(',').map(s => s.trim()).filter(Boolean);

            const productData = {
                name: document.getElementById('admin-p-name').value,
                price: Number(document.getElementById('admin-p-price').value),
                deposit: Number(document.getElementById('admin-p-deposit').value) || 0,
                desc: document.getElementById('admin-p-desc').value,

                    if (docId) {
                        await db.collection("products").doc(docId).update(productData);

                        productData.createdAt = firebase.firestore.FieldValue.serverTimestamp();

                btn.disabled = false;
                btn.innerText = "☁️ 儲存並發布至雲端資料庫";

        // 超強防護：完全剔除任何圖片數據，建立乾淨極輕量 (＜5KB) 訂單 payload
        async function submitOrder() {
            if (cart.length === 0) return alert("購物車目前空的喔！");
            
            const name = document.getElementById('buyer-name').value.trim();
            const igAccount = document.getElementById('buyer-ig').value.trim();
            const note = document.getElementById('buyer-note').value.trim();

            if (!name || !igAccount) {
                return alert("請完整填寫姓名與 IG 大帳帳號喔！");
            }

            const total = cart.reduce((sum, item) => sum + (item.price * item.qty), 0);
            const orderId = 'RZ' + Date.now().toString().slice(-6);

            // 嚴格淨化，絕對不含有圖片或 Base64 數據
            const strictCleanCart = cart.map(item => ({
                id: String(item.id || ''),
                title: String(item.name || ''),
                spec: String(item.specText || '標準'),
                price: Number(item.price || 0),
                qty: Number(item.qty || 1)
            }));

            let summaryText = `🛍️ rz_.market 訂單明細\n`;
            summaryText += `------------------\n`;
            summaryText += `訂單編號：#${orderId}\n`;
            summaryText += `訂購人：${name}\n`;
            summaryText += `IG 大帳：${igAccount}\n`;
            if (note) summaryText += `備註：${note}\n`;
            summaryText += `------------------\n`;
            summaryText += `商品明細：\n`;
            cart.forEach((item, index) => {
                summaryText += `${index + 1}. ${item.name} (${item.specText || '標準'}) x${item.qty} - NT$${item.price * item.qty}\n`;
            });
            summaryText += `------------------\n`;
            summaryText += `總金額：NT$ ${total}\n`;
            summaryText += `------------------\n`;
            summaryText += `請將此明細私訊傳給我們完成下單！✨`;

            try {
                if (isFirebaseConnected && db) {
                    await db.collection("orders").add({
                        orderId: orderId,
                        buyerName: name,
                        buyerIg: igAccount,
                        buyerNote: note,
                        items: strictCleanCart, // 絕對乾淨的輕量數據
                        total: total,
                        createdAt: firebase.firestore.FieldValue.serverTimestamp()
                    });
                }

                document.getElementById('order-summary-text').value = summaryText;
                copyOrderSummaryText(summaryText);

                cart = [];
                updateCartUI();
                toggleCartModal();
                document.getElementById('order-success-modal').classList.remove('hidden');
            } catch (err) {
                alert("下單錯誤：" + err.message);
            }
        }

        async function clearAllOrders() {
            if (!isAdminAuthenticated) return alert("請先驗證密碼！");
            if (confirm("確定要清空所有紀錄嗎？") && isFirebaseConnected && db) {
                const snapshot = await db.collection("orders").get();
                snapshot.forEach(doc => doc.ref.delete());
            }
        }

        async function deleteSingleOrder(id) {
            if (!isAdminAuthenticated) return alert("請先驗證密碼！");
            if (isFirebaseConnected && db) {
                await db.collection("orders").doc(id).delete();
            }
        }

        function handleAdminAccess() {
            if (isAdminAuthenticated) {
                toggleAdminModal();
            } else {
                document.getElementById('password-modal').classList.remove('hidden');
            }
        }

        function closePasswordModal() {
            document.getElementById('password-modal').classList.add('hidden');
            document.getElementById('admin-pass-input').value = "";
        }

        function verifyPassword() {
            const input = document.getElementById('admin-pass-input').value;
            if (input === ADMIN_SECRET) {
                isAdminAuthenticated = true;
                closePasswordModal();
                toggleAdminModal();
            } else {
                alert("❌ 密碼錯誤！無法進入專屬後台。");
            }
        }

        function toggleCartModal() {
            document.getElementById('cart-modal').classList.toggle('hidden');
        }
        function toggleAdminModal() {
            document.getElementById('admin-modal').classList.toggle('hidden');
        }
