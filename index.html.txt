        // 購物車邏輯
        function addToCart(id, specString = '') {
            const p = products.find(item => item.id === id);
            const itemKey = `${id}_${specString}`;
            const exist = cart.find(item => item.key === itemKey);

            if (exist) {
                exist.qty++;
            } else {
                cart.push({ ...p, key: itemKey, specText: specString, qty: 1 });
            }
            updateCartBadge();
            const specMsg = specString ? ` (${specString})` : '';
            alert(`已將「${p.name}${specMsg}」加入購物車！`);
        }

        function updateCartBadge() {
            const count = cart.reduce((sum, item) => sum + item.qty, 0);
            document.getElementById('cart-badge').textContent = count;
        }

        function openCart() {
            const container = document.getElementById('cart-items');
            container.innerHTML = '';
            let total = 0;
            let depositTotal = 0;

            if (cart.length === 0) {
                container.innerHTML = `<p class="text-xs text-center text-gray-400 py-10">購物車內無商品</p>`;

                cart.forEach((item, idx) => {
                    const sub = item.price * item.qty;
                    total += sub;
                    if (item.hasDeposit) {
                        depositTotal += item.depositPrice * item.qty;

                    container.innerHTML += `

                                <img src="${item.imgs[0] || ''}" class="w-10 h-10 object-cover rounded-lg">

                                </div>

        // 私訊結帳與訂單自動寫入後台
        function processCheckout(platform) {
            const name = document.getElementById('buyer-name').value.trim();
            const ig = document.getElementById('buyer-ig').value.trim();

            if (cart.length === 0) {
                alert('購物車是空的喔！');
                return;
            }

            if (!name || !ig) {
                alert('請填寫姓名與 IG 大帳，以便小幫手為您開立 7-11 賣貨便！');
                return;
            }

            let total = 0;
            let depositTotal = 0;
            let depositNotes = [];

            let detailText = `🛒 【rz_.market 官網訂購單】\n`;
            detailText += `👤 買家姓名：${name}\n`;
            detailText += `📱 IG 帳號：${ig}\n`;
            detailText += `--------------------\n`;

            const orderItems = [];
            cart.forEach(item => {
                const sub = item.price * item.qty;
                total += sub;
                if (item.hasDeposit) {
                    depositTotal += item.depositPrice * item.qty;
                    if (item.depositNote) depositNotes.push(`${item.name}: ${item.depositNote}`);
                }
                const specText = item.specText ? ` [${item.specText}]` : '';
                detailText += `▪️ ${item.name}${specText} x${item.qty} ($${sub})\n`;
                orderItems.push({ name: item.name, spec: item.specText, qty: item.qty, price: item.price });
            });

            detailText += `--------------------\n`;
            detailText += `💰 總計金額：$${total}\n`;
            if (depositTotal > 0) {
                detailText += `⚠️ 需預付訂金：$${depositTotal}\n`;
                if (depositNotes.length > 0) {
                    detailText += `📌 訂金說明：${depositNotes.join('; ')}\n`;
                }
            }
            detailText += `📦 寄送方式：7-11 賣貨便\n`;
            detailText += `✨ 請幫我開立賣貨便專屬連結，謝謝！`;

            // 自動儲存訂單至後台 (localStorage)
            const newOrder = {
                id: 'ORD-' + Date.now(),
                time: new Date().toLocaleString(),
                buyerName: name,
                buyerIG: ig,
                items: orderItems,
                total: total,
                depositTotal: depositTotal
            };
            orders.unshift(newOrder);
            saveStorage();

            // 複製到剪貼簿
            if (navigator.clipboard && window.isSecureContext) {
                navigator.clipboard.writeText(detailText);
            } else {
                const tempInput = document.createElement('textarea');
                tempInput.value = detailText;
                document.body.appendChild(tempInput);
                tempInput.select();
                document.execCommand('copy');

        // 上架新商品
        function addNewProduct() {
            const name = document.getElementById('new-p-name').value.trim();
            const price = parseFloat(document.getElementById('new-p-price').value);
            const type = document.getElementById('new-p-type').value;
            const desc = document.getElementById('new-p-desc').value.trim();
            const fileInput = document.getElementById('new-p-files');

            // 訂金
            const hasDeposit = document.getElementById('new-p-deposit-check').checked;
            const depositPrice = hasDeposit ? parseFloat(document.getElementById('new-p-deposit-price').value) || 0 : 0;
            const depositNote = hasDeposit ? document.getElementById('new-p-deposit-note').value.trim() : '';

            // 收集兩層規格 (主題 + 選項)
            const specGroups = [];
            document.querySelectorAll('.spec-group-box').forEach(box => {
                const title = box.querySelector('.spec-group-title').value.trim();
                const options = [];
                box.querySelectorAll('.spec-opt-value').forEach(opt => {
                    const val = opt.value.trim();
                    if (val) options.push(val);
                });
                if (title && options.length > 0) {
                    specGroups.push({ title, options });
                }
            });

            if (!name || isNaN(price)) {
                alert('請完整填寫商品名稱與價格！');
                return;
            }

            const imgList = [];
            const files = fileInput.files;

            if (files.length > 0) {
                let loaded = 0;
                for (let i = 0; i < files.length; i++) {
                    const reader = new FileReader();
                    reader.onload = function(evt) {
                        imgList.push(evt.target.result);
                        loaded++;
                        if (loaded === files.length) {
                            products.unshift({ id: Date.now(), name, price, type, imgs: imgList, desc, hasDeposit, depositPrice, depositNote, specGroups });
                            saveStorage();
                            renderProducts();
                            renderAdminProducts();
                            alert('商品上架成功！');
                            clearNewProductForm();
                        }
                    };
                    reader.readAsDataURL(files[i]);
                }
            } else {
                products.unshift({ id: Date.now(), name, price, type, imgs: ['https://images.unsplash.com/photo-1576995853123-5a10305d93c0?w=500&auto=format&fit=crop&q=60'], desc, hasDeposit, depositPrice, depositNote, specGroups });
                saveStorage();
                renderProducts();
                renderAdminProducts();
                alert('商品上架成功！(使用預設圖)');
                clearNewProductForm();
            }
        }

            });

            if (!name || isNaN(price)) {

            const fileInput = document.getElementById('edit-p-files');

                for (let i = 0; i < files.length; i++) {

            }
        }
